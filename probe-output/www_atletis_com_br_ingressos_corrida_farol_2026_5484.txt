<script>
    function updateKitLimits(kitId, kitMax) {
        if (!kitId || !kitMax) return;

        var $kitSelects = $('select[data-kit-id="' + kitId + '"]');
        if ($kitSelects.length <= 1) return;

        var totalSelected = 0;
        $kitSelects.each(function () {
            totalSelected += parseInt($(this).val()) || 0;
        });

        var remaining = kitMax - totalSelected;

        $kitSelects.each(function () {
            var $sel = $(this);
            var currentVal = parseInt($sel.val()) || 0;
            var maxForThis = currentVal + remaining;

            $sel.find('option').each(function () {
                var optVal = parseInt($(this).val()) || 0;
                var tooHigh = optVal > 0 && optVal > maxForThis;
                $(this).prop('disabled', tooHigh).toggle(!tooHigh);
            });
        });
    }

    function ticketChange() {
        $('select[name^="TicketsForm[tickets]"]')
            .off('change.kit')
            .on('change.kit', function () {
                var $select = $(this);
                var blockByAge = $select.data('block-age');

                if (blockByAge && $select.val() > 0) {
                    var modal = new bootstrap.Modal(document.getElementById('ageBlockModal'));
                    modal.show();
                }

                var kitId  = $select.data('kit-id');
                var kitMax = parseInt($select.data('kit-max')) || 0;
                updateKitLimits(kitId, kitMax);
            });
    }
</script>

    function ticketScript() {
        setModalTriggers();

    function buyTicket() {
        $(document).ready(function() {
            $('#form-tickets').on('submit', function(e) {
                e.preventDefault();
                $('#modal-buy-ticket').modal('show');
            });

            $('#btn-continue').on('click', function(e) {
                $('#form-tickets')[0].submit();
            });
        });
    }

    var setModalTriggers = function() {
        $("#terms-of-use-trigger").on('click', function() {
            $("#terms-of-use-iframe").attr('src', '/termos-de-uso');
        });
        $("#privacy-policy-trigger").on('click', function() {
            $("#privacy-policy-iframe").attr('src', '/politica-de-privacidade');
        });
    };

    coupon = function() {
        let $ticketCoupon = $('.ticket-coupon');

        $ticketCoupon.find('.insert-coupon').on('click', function() {
            let coupon = $ticketCoupon.find('input').val();
            $ticketCoupon.find('.error-block').remove();
            if (coupon) {
                return applyCoupon(coupon);
            }
            couponMessage('Digite o código do cupom!');
        });

        $('[data-toggle="remove-coupon"]').on('click', function() {
            removeCoupon();
        });
    };

    tickets = function() {
        $('.ticket-quantity select').on('change', function() {
            $('.total-tickets').text(toCurrency(getTotalCart()));

            controllerResume();
            revalideCoupon();
            handleButton();
        });
    };

    toCurrency = function(value) {
        return parseFloat(value).toLocaleString('pt-BR', {
            style: 'currency',
            currency: 'BRL',
            minimumFractionDigits: 2
        });
    };
    getTotalCart = function(withFee = true) {
        let $totalCart = 0;
        $('.ticket-quantity select').each(function() {
            if (Number($(this).val()) === 0) {
                return true;
            }

            let $code = $(this).data('code');
            let $price = getPrice($code)
            let $fee = getFee($code);

            if (withFee) {
                $totalCart += $(this).val() * ($price + $fee);
            } else {
                $totalCart += $(this).val() * $price;
            }
        });

        return $totalCart;
    };

    controllerResume = function() {
        const $resumePurchase = $('.ticket-resume-purchase');
        const $resumeTotal = $('.ticket-resume-total');
        const $formQuantity = $('.form-quantity');

        let resumeStr = '<span>Você selecionou para compra</span>';
        let total = 0;

        $formQuantity.each(function() {
            const $this = $(this);
            const quantity = parseInt($this.val());

            if (quantity < 1) {
                return true;
            }

            const code = $this.data('code');
            const title = $this.data('title');
            const price = getPrice(code);
            const fee = getFee(code);
            const totalPrice = quantity * (price + fee);

            resumeStr += `<span>${quantity} ingresso${quantity === 1 ? '' : 's'} ${title} ${toCurrency(price + fee)}</span>`;
            total += totalPrice;
        });

        $resumePurchase.html(resumeStr);
        $resumeTotal.html(`<span>Valor total</span><span class="total-tickets">${toCurrency(total)}</span>`);
    };

    getPrice = function(code) {
        return parseFloat($('.ticket-price[data-code="' + code + '"]').data('price'));
    };
    getFee = function(code) {
        return parseFloat($('.ticket-price[data-code="' + code + '"]').data('fee'));
    };
    applyCoupon = function(coupon) {
        let $ticketCoupon = $('.ticket-coupon');
        let $totalCart = Number(getTotalCart(false));

        let $couponDiscount = $('.coupon-discount');

        let tickets = {};
        $('.ticket-quantity select').each(function() {
            tickets[$(this).data('code')] = $(this).val();
        });

        $.ajax({
            type: "POST",
            url: '/check-coupon',
            data: {
                '_csrf-app': 'l2YQE7iIMJKy6WRndWdayfHKFpliP-UfNJDIl2ox22ClLHFK99EF_4qsE1MGCjato6Vc7RJqyCpV5ZDmDUiOEQ==',
                'coupon_code': coupon,
                'event_id': 7323,
                'tickets': tickets,
                'value_cart': $totalCart
            },
            beforeSend: function() {
                $("#has-coupon").hide();
            },
            success: function(data) {
                if (data.success) {
                    let $couponDiscount = $(".coupon-discount");

                    $couponDiscount
                        .removeClass('d-none')
                        .find('#coupon-value')
                        .text('-' + toCurrency(data.discount_value_granted));

                    $couponDiscount
                        .removeClass('d-none')
                        .find('#subtotal-value')
                        .text(toCurrency(getTotalCart()));

                    $('.ticket-resume-total > span:nth-of-type(2)').text(
                        toCurrency(getTotalCart() - data.discount_value_granted)
                    );

                    if ($('a.coupon-toggler.collapsed').length === 0) {
                        $ticketCoupon.find('.coupon-toggler').text(coupon).trigger('click');
                    }

                    $couponDiscount.attr('data-coupon-amount', data.discount_value_granted);
                    $couponDiscount.attr('data-coupon-code', coupon);

                        $('#coupon-code').val(coupon);
                        $('#coupon-amount').val(data.discount_value_granted);
                        $('span#coupon-name').text(coupon);
                    } else {
                        $ticketCoupon
                            .find('.form-group:nth-of-type(1)')
                            .addClass('has-error');
                        if ($ticketCoupon.find('.error-block').length === 0) {
                            $ticketCoupon
                                .find('.form-group:nth-of-type(1)')
                                .append('<div class="error-block">' +
                                    data.discount_message +
                                    '</div>'
                                );
                        } else {
                            $ticketCoupon.find('.error-block').text(data.discount_message);
                        }
                        
                        removeCoupon();
                        $('.clear-coupon').addClass('d-none');
                        $couponDiscount.addClass('d-none');
                        $('#coupon-field').collapse('show');
                        $ticketCoupon.find('.coupon-toggler').text('Tem cupom de desconto?');
                        $couponDiscount.attr('data-coupon-code', '0');
                    }
                    $("#has-coupon").show();
                },
                error: function () {
                    removeCoupon();
                    $("#has-coupon").show();
                    $couponDiscount.addClass('d-none');
                    $('#coupon-field').collapse('hide');
                    $ticketCoupon.find('.coupon-toggler').text('Tem cupom de desconto?');
                    $couponDiscount.attr('data-coupon-code', '0');
                }
            });
        };
        revalideCoupon = function () {
            let $codeCoupon = ($('.coupon-discount').attr('data-coupon-code')).trim();
            if ($codeCoupon) {
                applyCoupon($codeCoupon);
                return true;
            }
            $('#coupon-field').collapse('hide');
        };
        validationOnSubmit = function () {
            $('#form-tickets').on('submit', function (e) {
                let hasQuantity = false;
                let $modal = $('#modalMessages');
                let $modalBody = $modal.find('.modal-body');
                let $coupon = $('#coupon');
                let $message = '';

            $('.ticket-quantity select').each(function() {
                if ($(this).val() > 0) {
                    hasQuantity = true;
                    return true;
                }
            });

            if (!hasQuantity) {
                $message +=
                    '<li>Escolha pelo menos um ingresso. Não identificamos nenhum ingresso no carrinho!</li>';
            }

            if ($message !== '') {
                $modalBody.html($message);
                $modal.modal('show');
                e.preventDefault();
                return false;
            }

            if (($coupon.is(':visible') && $coupon.val()) || $coupon.is(':focus')) {
                e.preventDefault();
                $coupon.siblings('button').trigger('click');
            }

            $items = [];
            $('.ticket-quantity select').each(function() {
                if ($(this).val() > 0) {
                    const code = $(this).data('code');
                    const title = $(this).data('title');
                    const price = getPrice(code);
                    const fee = getFee(code);
                    const totalPrice = price + fee;

                    $items.push({
                        item_id: code,
                        item_name: title,
                        price: totalPrice,
                        quantity: $(this).val()
                    });
                }
            });

            const dataLayerObject = {
                'event': 'begin_checkout',
                'ecommerce': {
                    'currency': 'BRL',
                    'value': getTotalCart().toFixed(2),
                    'coupon': ($('.coupon-discount').attr('data-coupon-code')).trim(),
                    'items': $items
                }
            };

            dataLayer.push(dataLayerObject);
        });
    };
